const menuItems = document.querySelectorAll('.menu-item input[type="checkbox"]'); const subtotalValue = document.getElementById('subtotal-value'); let subtotal = 0; // Función para calcular el subtotal function calcularSubtotal() { subtotal = 0; menuItems.forEach(item => { if (item.checked) { subtotal += parseFloat(item.dataset.price); } }); subtotalValue.textContent = subtotal.toFixed(2); } // Agregamos un listener a cada checkbox menuItems.forEach(item => { item.addEventListener('change', calcularSubtotal); }); // Inicializamos el subtotal calcularSubtotal();
